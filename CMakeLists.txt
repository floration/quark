# The version lives in one generated .cpp (src/common/version.h), not a global define: a bump recompiles one file.
# QUARK_VERSION_SUFFIX: testing builds are <VERSION>.testing.<N> (release.yml, from 1.19.0 on): VERSION stays at the
# production release they follow, and they sort above it (and below the next one). Before: <VERSION>-test.<N>.
set(QUARK_VERSION_SUFFIX "" CACHE STRING "Appended to VERSION (CI testing builds: .testing.<run number>)")
file(STRINGS ${CMAKE_SOURCE_DIR}/VERSION QUARK_VERSION LIMIT_COUNT 1)
string(STRIP "${QUARK_VERSION}${QUARK_VERSION_SUFFIX}" QUARK_VERSION)
if(NOT QUARK_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+(\\.testing\\.[0-9]+|-[0-9A-Za-z.]+)?$")
  message(FATAL_ERROR "version '${QUARK_VERSION}' isn't x.y.z[.testing.N|-suffix]")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/VERSION)
# (the launcher marker is found in launcher files by src/launcher/selfupdate.cpp, so it's one contiguous string)
file(WRITE ${CMAKE_BINARY_DIR}/generated/version.cpp.tmp
  "// generated by CMake from VERSION\n#include \"common/version.h\"\n"
  "const char kQuarkVersion[] = \"${QUARK_VERSION}\";\n"
  "const char kQuarkLauncherMarker[] = \"QUARK-LAUNCHER-VERSION:${QUARK_VERSION}\";\n")
configure_file(${CMAKE_BINARY_DIR}/generated/version.cpp.tmp ${CMAKE_BINARY_DIR}/generated/version.cpp COPYONLY)

# Windows resources for every shipped exe (src/win/app.rc.in, app.manifest.in): the icon, a VERSIONINFO (product quark,
# company floration, a description, the version incl. a .testing.N (or -test.N) suffix: FILEVERSION x.y.z.N, flagged prerelease) and an
# application manifest (asInvoker, Windows 7-11, per-monitor DPI for the windowed ones, as GLFW sets at runtime).
# Anonymous exes without them are what antivirus heuristics flag most (docs/SIGNING.md). Exes are never packed.
if(WIN32)
  string(REGEX MATCH "^([0-9]+)\\.([0-9]+)\\.([0-9]+)" _qv "${QUARK_VERSION}")
  set(QV_MAJOR ${CMAKE_MATCH_1})
  set(QV_MINOR ${CMAKE_MATCH_2})
  set(QV_PATCH ${CMAKE_MATCH_3})
  set(QV_BUILD 0)
  set(QV_FLAGS 0x0L)
  if(QUARK_VERSION MATCHES "-|\\.testing\\.")
    set(QV_FLAGS 0x2L)   # VS_FF_PRERELEASE
    if(QUARK_VERSION MATCHES "(-test|\\.testing)\\.([0-9]+)$")
      set(QV_BUILD ${CMAKE_MATCH_2})
      if(QV_BUILD GREATER 65535)
        set(QV_BUILD 65535)
      endif()
    endif()
  endif()
  function(quark_win_resources target file desc windowed)
    set(QUARK_RES_NAME ${target})
    set(QUARK_RES_FILE ${file})
    set(QUARK_RES_DESC "${desc}")
    set(QUARK_RES_ICON ${CMAKE_SOURCE_DIR}/assets/quark.ico)
    set(QUARK_RES_MANIFEST ${CMAKE_BINARY_DIR}/generated/${target}.manifest)
    set(QUARK_RES_DPI "")
    if(windowed)
      set(QUARK_RES_DPI "  <application xmlns=\"urn:schemas-microsoft-com:asm.v3\">\n    <windowsSettings>\n      <dpiAware xmlns=\"http://schemas.microsoft.com/SMI/2005/WindowsSettings\">true/pm</dpiAware>\n      <dpiAwareness xmlns=\"http://schemas.microsoft.com/SMI/2016/WindowsSettings\">PerMonitorV2, PerMonitor</dpiAwareness>\n    </windowsSettings>\n  </application>\n")
    endif()
    configure_file(${CMAKE_SOURCE_DIR}/src/win/app.manifest.in ${QUARK_RES_MANIFEST} @ONLY)
    configure_file(${CMAKE_SOURCE_DIR}/src/win/app.rc.in ${CMAKE_BINARY_DIR}/generated/${target}.rc @ONLY)
    target_sources(${target} PRIVATE ${CMAKE_BINARY_DIR}/generated/${target}.rc)
    set_source_files_properties(${CMAKE_BINARY_DIR}/generated/${target}.rc PROPERTIES
      OBJECT_DEPENDS "${QUARK_RES_MANIFEST};${QUARK_RES_ICON}")
  endfunction()
endif()

# Robustness builds (REVIEW ENG-3), never for releases:
#   QUARK_SANITIZE: everything built with AddressSanitizer + UndefinedBehaviorSanitizer (the CI "sanitizers" job runs
#                   quark_tests like that); any report is fatal.
#   QUARK_FUZZ:     libFuzzer harnesses (tests/fuzz/fuzz_*.cpp) for what reads untrusted bytes: packets (client
#                   decode, and a live server's packets and string commands), Map::parse, CustomRules::fromText, the
#                   launcher's .qpak reader. clang only (Ubuntu: clang + libclang-rt-18-dev); implies QUARK_SANITIZE.
#                     CC=clang CXX=clang++ cmake -S . -B build-fuzz -DQUARK_FUZZ=ON -DQUARK_BUILD_CLIENT=OFF \
#                       -DCMAKE_BUILD_TYPE=RelWithDebInfo && cmake --build build-fuzz -j
#                     cd build-fuzz && mkdir -p corpus && cp ../tests/data/fuzz/map/* corpus/ && ./fuzz_map corpus
#                   (fuzz_server needs to run in the build folder: it loads maps/bastion.qmap). Seeds: tests/data/fuzz/
#                   <harness>/; put a crash input there after fixing it, so the CI replay keeps checking it.
option(QUARK_SANITIZE "Build with AddressSanitizer + UndefinedBehaviorSanitizer" OFF)
option(QUARK_FUZZ "Build the libFuzzer harnesses (clang; implies QUARK_SANITIZE)" OFF)
if(QUARK_LTO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT QUARK_IPO_OK OUTPUT QUARK_IPO_ERR LANGUAGES C CXX)

    set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
  else()
    message(WARNING "LTO not supported here: ${QUARK_IPO_ERR}")
  endif()
endif()

# toolchain bit for bit, so no fast-math and no FMA contraction

  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "QUARK_FUZZ needs clang (libFuzzer): CC=clang CXX=clang++ cmake ... -DQUARK_FUZZ=ON")
  endif()
  set(QUARK_SANITIZE ON)
  add_compile_options(-fsanitize=fuzzer-no-link)   # coverage for the fuzzers in every library they link
endif()
if(QUARK_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=address,undefined)
  add_compile_definitions(QUARK_SANITIZED=1)   # (quark_tests skips a pure timing check: sanitized code is slower)
endif()

# ---- maps (a build artifact) ---------------------------------------------------------------------------
# tools/make_maps.py generates the shipped maps at build time (python3, nothing else) into maps/, which .gitignore
# (the generator: tools/maps/prefabs.py, the shared builder, and tools/maps/<name>.py, one module per map)
# leaves out: the generator is the only source. maps/ is where the game looks (next to the source tree when run from
# it, copied next to the executables below) and what release packaging copies. A hand-made map ships (embedded and
# copied) only when .gitignore un-ignores it with a "!/maps/<name>.qmap" line, i.e. it's tracked: any other *.qmap
# in maps/ (one the generator no longer makes, a test leftover, a stray file) is neither embedded nor copied.
find_package(Python3 REQUIRED COMPONENTS Interpreter)
# The built-in bot brain (tools/bots/default.qbrain, tools/bots/GRAMMAR.md) is embedded at build time.
add_custom_command(OUTPUT ${CMAKE_BINARY_DIR}/generated/default_brain.h
  COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/bots/embed_brain.py ${CMAKE_BINARY_DIR}/generated ${CMAKE_SOURCE_DIR}/tools/bots/default.qbrain
  DEPENDS ${CMAKE_SOURCE_DIR}/tools/bots/embed_brain.py ${CMAKE_SOURCE_DIR}/tools/bots/default.qbrain
  COMMENT "Embedding the built-in bot brain (tools/bots/default.qbrain)" VERBATIM)
# The built-in modes' logic (tools/modes/common.qlogic + one .qlogic per mode, tools/modes/GRAMMAR.md) likewise.
set(QUARK_MODE_LOGIC_FILES common vs bomb race koth payload ffa npayload duel arena tourney hide tag points)
list(TRANSFORM QUARK_MODE_LOGIC_FILES PREPEND ${CMAKE_SOURCE_DIR}/tools/modes/)
list(TRANSFORM QUARK_MODE_LOGIC_FILES APPEND .qlogic)
add_custom_command(OUTPUT ${CMAKE_BINARY_DIR}/generated/default_modes.h
  COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/modes/embed_modes.py ${CMAKE_BINARY_DIR}/generated ${CMAKE_SOURCE_DIR}/tools/modes
  DEPENDS ${CMAKE_SOURCE_DIR}/tools/modes/embed_modes.py ${QUARK_MODE_LOGIC_FILES}
  COMMENT "Embedding the built-in modes' logic (tools/modes/*.qlogic)" VERBATIM)
set(QUARK_MAPS_DIR ${CMAKE_SOURCE_DIR}/maps)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/tools/make_maps.py)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/make_maps.py --list
  OUTPUT_VARIABLE QUARK_GEN_MAP_NAMES RESULT_VARIABLE rc OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT rc EQUAL 0 OR QUARK_GEN_MAP_NAMES STREQUAL "")
  message(FATAL_ERROR "tools/make_maps.py --list failed (${rc})")
endif()
string(REPLACE "\r" "" QUARK_GEN_MAP_NAMES "${QUARK_GEN_MAP_NAMES}")
string(REPLACE "\n" ";" QUARK_GEN_MAP_NAMES "${QUARK_GEN_MAP_NAMES}")
set(QUARK_GEN_MAPS "")
foreach(n ${QUARK_GEN_MAP_NAMES})
  list(APPEND QUARK_GEN_MAPS ${QUARK_MAPS_DIR}/${n}.qmap)
endforeach()
# (the owner's map overrides, assets/overrides/maps/<name>.qmap, replace generated maps: docs/ASSETS.md)
file(GLOB QUARK_MAP_OVERRIDES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/assets/overrides/maps/*.qmap)
# (any change to the generator, the shared builder or one map's module, regenerates; a new module re-globs)
file(GLOB QUARK_MAP_SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/tools/maps/*.py)
add_custom_command(OUTPUT ${QUARK_GEN_MAPS}
  COMMAND ${CMAKE_COMMAND} -E env PYTHONHASHSEED=0 ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/make_maps.py --out ${QUARK_MAPS_DIR}
  DEPENDS ${CMAKE_SOURCE_DIR}/tools/make_maps.py ${QUARK_MAP_SOURCES} ${QUARK_MAP_OVERRIDES}
  COMMENT "Generating maps (tools/make_maps.py)" VERBATIM)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/.gitignore)
file(STRINGS ${CMAKE_SOURCE_DIR}/.gitignore QUARK_HAND_MAP_LINES REGEX "^!/maps/[A-Za-z0-9_-]+\\.qmap")
set(QUARK_HAND_MAPS "")
foreach(l ${QUARK_HAND_MAP_LINES})
  string(REGEX REPLACE "^!/maps/([A-Za-z0-9_-]+)\\.qmap.*$" "\\1" n "${l}")
  if(n IN_LIST QUARK_GEN_MAP_NAMES)
    continue()
  elseif(EXISTS ${QUARK_MAPS_DIR}/${n}.qmap)
    list(APPEND QUARK_HAND_MAPS ${QUARK_MAPS_DIR}/${n}.qmap)
  else()
    message(WARNING ".gitignore ships maps/${n}.qmap, but there's no such file")
  endif()
endforeach()

# Copy the shipped maps (generated + hand-made, as embedded above) next to the executables so builds run in place;
# any other *.qmap there (a map since retired from the generator, a stray file) is removed.
string(REPLACE ";" "|" QUARK_SHIPPED_MAPS_ARG "${QUARK_GEN_MAPS};${QUARK_HAND_MAPS}")
add_custom_target(copy_maps ALL
  COMMAND ${CMAKE_COMMAND} "-DMAPS=${QUARK_SHIPPED_MAPS_ARG}" -DDEST=${CMAKE_BINARY_DIR}/maps -P ${CMAKE_SOURCE_DIR}/tools/copy_maps.cmake
  VERBATIM)
add_dependencies(copy_maps quark_maps)

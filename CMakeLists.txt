# The version lives in one generated .cpp (src/common/version.h), not a global define: a bump recompiles one file.
# QUARK_VERSION_SUFFIX: testing builds are <VERSION>-test.<N> (release.yml), so they sort below the production release.
set(QUARK_VERSION_SUFFIX "" CACHE STRING "Appended to VERSION (CI testing builds: -test.<run number>)")
file(STRINGS ${CMAKE_SOURCE_DIR}/VERSION QUARK_VERSION LIMIT_COUNT 1)
string(STRIP "${QUARK_VERSION}${QUARK_VERSION_SUFFIX}" QUARK_VERSION)
if(NOT QUARK_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+(-[0-9A-Za-z.]+)?$")
  message(FATAL_ERROR "version '${QUARK_VERSION}' isn't x.y.z[-suffix]")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/VERSION)
# (the launcher marker is found in launcher files by src/launcher/selfupdate.cpp, so it's one contiguous string)
file(WRITE ${CMAKE_BINARY_DIR}/generated/version.cpp.tmp
  "// generated by CMake from VERSION\n#include \"common/version.h\"\n"
  "const char kQuarkVersion[] = \"${QUARK_VERSION}\";\n"
  "const char kQuarkLauncherMarker[] = \"QUARK-LAUNCHER-VERSION:${QUARK_VERSION}\";\n")
configure_file(${CMAKE_BINARY_DIR}/generated/version.cpp.tmp ${CMAKE_BINARY_DIR}/generated/version.cpp COPYONLY)

# Official maps (ranked only runs on these): name + sha256 of each shipped maps/*.qmap, line endings
# normalized so a CRLF checkout hashes the same. Re-generated whenever a map file changes.
file(GLOB QUARK_MAP_FILES ${CMAKE_SOURCE_DIR}/maps/*.qmap)
set(QUARK_OFFICIAL_MAPS "")
foreach(f ${QUARK_MAP_FILES})
  get_filename_component(n ${f} NAME_WE)
  if(NOT n MATCHES "^zz_")
    file(READ ${f} txt)
    string(REPLACE "\r" "" txt "${txt}")
    string(SHA256 h "${txt}")
    string(APPEND QUARK_OFFICIAL_MAPS "    {\"${n}\", \"${h}\"},\n")
    # the shipped text too (CR stripped), so the game can put a map back to its original version
    file(READ ${f} hex HEX)
    string(REGEX REPLACE "(..)" "0x\\1," hex "${hex}")
    string(REPLACE "0x0d," "" hex "${hex}")
    string(APPEND QUARK_MAP_DATA "static const unsigned char kMapData_${n}[] = {${hex}};\n")
    string(APPEND QUARK_MAP_TABLE "    {\"${n}\", kMapData_${n}, sizeof kMapData_${n}},\n")
  endif()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${f})
endforeach()
file(WRITE ${CMAKE_BINARY_DIR}/generated/official_maps.h.tmp
  "// generated by CMake from maps/*.qmap\n#pragma once\nstruct OfficialMap { const char* name; const char* sha256; };\n"
  "static const OfficialMap kOfficialMaps[] = {\n${QUARK_OFFICIAL_MAPS}    {nullptr, nullptr}};\n")
configure_file(${CMAKE_BINARY_DIR}/generated/official_maps.h.tmp ${CMAKE_BINARY_DIR}/generated/official_maps.h COPYONLY)
file(WRITE ${CMAKE_BINARY_DIR}/generated/official_map_data.h.tmp
  "// generated by CMake from maps/*.qmap: the original text of every shipped map\n#pragma once\n#include <cstddef>\n${QUARK_MAP_DATA}"
  "struct OfficialMapData { const char* name; const unsigned char* data; size_t size; };\n"
  "static const OfficialMapData kOfficialMapData[] = {\n${QUARK_MAP_TABLE}    {nullptr, nullptr, 0}};\n")
configure_file(${CMAKE_BINARY_DIR}/generated/official_map_data.h.tmp ${CMAKE_BINARY_DIR}/generated/official_map_data.h COPYONLY)
# Profiling: -DQUARK_TRACY=ON adds Tracy zones (src/common/profile.h). QUARK_TRACY_DIR can point at a
# local Tracy checkout; otherwise v0.11.1 is fetched. Off by default (release builds never include it).

  add_compile_definitions(QUARK_ZONESTATS)

if(QUARK_TRACY)
  if(QUARK_TRACY_DIR)
    set(tracy_SOURCE_DIR ${QUARK_TRACY_DIR})
  else()
    include(FetchContent)
    FetchContent_Declare(tracy GIT_REPOSITORY https://github.com/wolfpld/tracy.git GIT_TAG v0.11.1 GIT_SHALLOW TRUE)
    FetchContent_Populate(tracy)
  endif()
  target_sources(quark_core PRIVATE ${tracy_SOURCE_DIR}/public/TracyClient.cpp)
  target_include_directories(quark_core PUBLIC ${tracy_SOURCE_DIR}/public)
  target_compile_definitions(quark_core PUBLIC QUARK_TRACY TRACY_ENABLE)
  target_link_libraries(quark_core PUBLIC ${CMAKE_DL_LIBS})
endif()

if(QUARK_BUILD_CLIENT)
  set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
  add_subdirectory(third_party/glfw EXCLUDE_FROM_ALL)
  target_compile_options(glfw PRIVATE -Wno-sign-compare -Wno-missing-field-initializers)   # third party: our -Wextra is for our code

  # The recorded announcer lines (20 MB of samples) are generated from assets/voice/ at build time, not committed
  # (tools/import_voice.py, python3 standard library only).
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  file(GLOB_RECURSE QUARK_VOICE_WAVS ${CMAKE_SOURCE_DIR}/assets/voice/*.wav)
  add_custom_command(
    OUTPUT ${CMAKE_BINARY_DIR}/generated/voice_recorded.h
    COMMAND Python3::Interpreter ${CMAKE_SOURCE_DIR}/tools/import_voice.py ${CMAKE_BINARY_DIR}/generated/voice_recorded.h
    DEPENDS ${CMAKE_SOURCE_DIR}/tools/import_voice.py ${CMAKE_SOURCE_DIR}/tools/make_voice.py ${QUARK_VOICE_WAVS}
    COMMENT "Generating voice_recorded.h from assets/voice/"
    VERBATIM)

  add_executable(quark WIN32 MACOSX_BUNDLE
    src/client/main.cpp
    src/client/game.cpp
    src/client/community.cpp
    src/client/editor.cpp
    src/client/openurl.cpp
    src/client/gl.cpp
    src/client/renderer.cpp
    src/client/lightmap.cpp
    src/client/ui.cpp
    src/client/config.cpp
    src/client/hudwidgets.cpp
    src/client/profiles.cpp
    src/client/binds.cpp
    src/client/audio.cpp
    ${CMAKE_BINARY_DIR}/generated/voice_recorded.h
    ${QUARK_ICON_RC}
  )
  target_link_libraries(quark quark_core quark_launcher_core glfw ${CMAKE_DL_LIBS})
  if(WIN32)
    target_link_libraries(quark shell32)
  endif()
  add_executable(quark_launcher WIN32
    src/launcher/main.cpp
    ${QUARK_ICON_RC}
    src/client/gl.cpp
    src/client/renderer.cpp
    src/client/lightmap.cpp
    src/client/ui.cpp
  )
  target_link_libraries(quark_launcher quark_core quark_launcher_core glfw)
  foreach(t quark quark_launcher)
    if(APPLE)
      target_link_libraries(${t} "-framework OpenGL" "-framework AudioToolbox" "-framework CoreFoundation")
    elseif(WIN32)
      target_link_libraries(${t} opengl32)
    endif()
  endforeach()
  if(WIN32)
    target_link_libraries(quark_launcher winhttp ole32 uuid shell32 advapi32)
  endif()
endif()
